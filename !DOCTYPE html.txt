<script>
const codonTable={
UUU:"フェニルアラニン",UUC:"フェニルアラニン",UUA:"ロイシン",UUG:"ロイシン",
UCU:"セリン",UCC:"セリン",UCA:"セリン",UCG:"セリン",
UAU:"チロシン",UAC:"チロシン",UAA:"",UAG:"",
UGU:"システイン",UGC:"システイン",UGA:"",UGG:"トリプトファン",
CUU:"ロイシン",CUC:"ロイシン",CUA:"ロイシン",CUG:"ロイシン",
CCU:"プロリン",CCC:"プロリン",CCA:"プロリン",CCG:"プロリン",
CAU:"ヒスチジン",CAC:"ヒスチジン",CAA:"グルタミン",CAG:"グルタミン",
CGU:"アルギニン",CGC:"アルギニン",CGA:"アルギニン",CGG:"アルギニン",
AUU:"イソロイシン",AUC:"イソロイシン",AUA:"イソロイシン",AUG:"メチオニン",
ACU:"トレオニン",ACC:"トレオニン",ACA:"トレオニン",ACG:"トレオニン",
AAU:"アスパラギン",AAC:"アスパラギン",AAA:"リシン",AAG:"リシン",
AGU:"セリン",AGC:"セリン",AGA:"アルギニン",AGG:"アルギニン",
GUU:"バリン",GUC:"バリン",GUA:"バリン",GUG:"バリン",
GCU:"アラニン",GCC:"アラニン",GCA:"アラニン",GCG:"アラニン",
GAU:"アスパラギン酸",GAC:"アスパラギン酸",GAA:"グルタミン酸",GAG:"グルタミン酸",
GGU:"グリシン",GGC:"グリシン",GGA:"グリシン",GGG:"グリシン"
};
function escapeHtml(text){
return text.replace(/[&<>"']/g,function(c){
return{"&":"&amp;","<":"&lt;",">":"&gt;","\"":"&quot;","'":"&#39;"}[c];
});
}
function countBase(rna,base){
return (rna.match(new RegExp(base,"g"))||[]).length;
}
function analyze(input){
const rna=input.toUpperCase().replace(/\s/g,"");
if(rna.length===0){
return{type:"error",message:"A、U、G、Cを入力してください",counts:null,aminoAcids:[],aminoAcidCount:0};
}
if(!/^[AUGC]+$/.test(rna)){
return{type:"error",message:"A、U、G、Cだけを入力してください",counts:null,aminoAcids:[],aminoAcidCount:0};
}
const counts={
A:countBase(rna,"A"),
U:countBase(rna,"U"),
G:countBase(rna,"G"),
C:countBase(rna,"C")
};
const begin=rna.indexOf("AUG");
if(begin===-1){
return{type:"message",message:"どこからアミノ酸を合成したらいいかわかりません",counts:counts,aminoAcids:[],aminoAcidCount:0};
}
const aminoAcids=[];
for(let i=begin;i<=rna.length-3;i+=3){
const codon=rna.slice(i,i+3);
if(codon==="UAA"||codon==="UAG"||codon==="UGA")break;
aminoAcids.push(codonTable[codon]);
}
return{type:"success",message:"",counts:counts,aminoAcids:aminoAcids,aminoAcidCount:aminoAcids.length};
}
function render(result){
const resultBox=document.getElementById("result");
let html="";
if(result.type==="error"){
html+=`<div class="section"><p class="error">${escapeHtml(result.message)}</p></div>`;
}else if(result.type==="message"){
html+=`<div class="section"><p class="message">${escapeHtml(result.message)}</p></div>`;
}else{
const list=result.aminoAcids.length>0?escapeHtml(result.aminoAcids.join(" - ")):"";
html+=`<div class="section"><h2 class="section-title">アミノ酸配列</h2><div class="amino-list">${list}</div></div>`;
}
if(result.counts){
html+=`<div class="section"><h2 class="section-title">塩基の数</h2><div class="stats">`;
html+=`<div class="stat">A:<span class="count">${result.counts.A}</span>個</div>`;
html+=`<div class="stat">U:<span class="count">${result.counts.U}</span>個</div>`;
html+=`<div class="stat">G:<span class="count">${result.counts.G}</span>個</div>`;
html+=`<div class="stat">C:<span class="count">${result.counts.C}</span>個</div>`;
html+=`</div></div>`;
}
html+=`<div class="section"><h2 class="section-title">合成されたアミノ酸の数</h2><div class="stat">${result.aminoAcidCount}個</div></div>`;
resultBox.innerHTML=html;
resultBox.classList.add("show");
}
document.getElementById("runBtn").addEventListener("click",function(){
render(analyze(document.getElementById("rnaInput").value));
});
document.getElementById("resetBtn").addEventListener("click",function(){
document.getElementById("rnaInput").value="";
const resultBox=document.getElementById("result");
resultBox.innerHTML="";
resultBox.classList.remove("show");
});
document.getElementById("rnaInput").addEventListener("input",function(){
this.value=this.value.toUpperCase();
});
